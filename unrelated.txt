FROM docker-ng-releases.northgrum.com/redhat/ubi9/python-312:9.5-1743509868

USER root

# Add NG certs
RUN curl http://certdata.northropgrumman.com/certdata/certs/NorthropGrummanCorporateRootCA-384.cer \
    -o /etc/pki/ca-trust/source/anchors/NorthropGrummanCorporateRootCA-384.cer

RUN curl http://certdata.northropgrumman.com/certdata/certs/NorthropGrummanMultiPurposeCA-G2.cer \
    -o /etc/pki/ca-trust/source/anchors/NorthropGrummanMultiPurposeCA-G2.cer

RUN update-ca-trust

WORKDIR /app

# Internal repos
COPY nexus.repo /etc/yum.repos.d/
COPY microsoft-proxy.repo /etc/yum.repos.d/

# Avoid entitlement issues
RUN rm -f /etc/yum.repos.d/ubi.repo
RUN rm -f /etc/yum.repos.d/redhat.repo

# Prevent extra dependency bloat
RUN echo "install_weak_deps=False" >> /etc/yum.conf

# Install SQL Server ODBC driver
RUN ACCEPT_EULA=Y yum install -y msodbcsql18

RUN yum clean all

RUN rm -rf /var/cache/yum

# Python setup
COPY requirements.txt .
COPY pip.conf .
COPY NG-Certificate-Chain.cer .

ENV PIP_CONFIG_FILE=/app/pip.conf

RUN pip install --no-cache-dir -r requirements.txt

# If pyodbc compilation fails, uncomment this and rebuild:
# RUN yum install -y unixODBC-devel

# Copy application
COPY . .

EXPOSE 8501

CMD ["streamlit", "run", "entry.py", "--server.port=8501", "--server.address=0.0.0.0"]
